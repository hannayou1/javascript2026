문제 1. 탭 메뉴 HTML 작성형

배점 30점 · 권장 시간 8분

다음 JS가 정상 동작하도록 HTML 마크업을 작성하시오.
조건: 첫 번째 탭이 기본 활성화 상태, 탭은 총 3개 — "공지사항", "이벤트", "FAQ"

주어진 JS

const tabBtns = document.querySelectorAll('.tab-btn');
const tabContents = document.querySelectorAll('.tab-content');

tabBtns.forEach(function (btn) {
  btn.addEventListener('click', function () {
    tabBtns.forEach(b => b.classList.remove('active'));
    tabContents.forEach(c => c.classList.remove('active'));

    this.classList.add('active');

    // data-tab 값과 같은 id를 가진 컨텐츠 활성화
    const target = document.getElementById(this.dataset.tab);
    target.classList.add('active');
  });
});

  • 클래스명은 JS 셀렉터와 정확히 일치: tab-btn, tab-content
  • this.dataset.tab → 버튼에 data-tab 속성 필요
  • getElementById(this.dataset.tab) → data-tab 값 = 컨텐츠 id 일치 (최다 오답 포인트)
  • 첫 버튼·첫 컨텐츠에 active 미리 부여
  • type="button"은 가산점 (form 안 submit 방지)

const titles = document.querySelectorAll('.acc-title');

titles.forEach(function (title) {

문제 3. 체크박스 전체 선택 오류 수정

배점 40점 · 권장 시간 10분

아래 코드는 "전체 선택" 기능인데 오류가 4곳 있다. 모두 찾아 수정하시오.
동작: 전체선택 클릭 시 하위 체크박스 모두 체크/해제, 하위를 하나라도 해제하면 전체선택도 해제.

주어진 HTML

<label><input type="checkbox" id="checkAll"> 전체 선택</label>
<ul>
  <li><label><input type="checkbox" class="chk"> 항목 1</label></li>
  <li><label><input type="checkbox" class="chk"> 항목 2</label></li>
  <li><label><input type="checkbox" class="chk"> 항목 3</label></li>
</ul>

오류가 포함된 JS

const checkAll = document.getElementByID('checkAll');
const checks = document.querySelectorAll('.chk');

checkAll.addEventListener('change', function () {
  checks.forEach(function (chk) {
    chk.checked = checkAll.value;
  });
});

checks.forEach(function (chk) {
  chk.addEventListener('change', function () {
    let allChecked = true;
    checks.forEach(function (c) {
      if (c.checked = false) {
        allChecked = false;
      }
    });
    checkAll.checked = allchecked;
  });
});

▼ 완성 시 동작 예시 (직접 클릭해보세요)

const checkAll = document.getElementById('checkAll');   // (1)

  4. allchecked → allChecked — 변수명 대소문자 불일치(ReferenceError)

문제 4. 이미지 슬라이더 빈칸 채우기

배점 30점 · 권장 시간 8분

이전/다음 버튼으로 슬라이드가 넘어가고, 마지막에서 다음을 누르면 처음으로 순환하도록 ①~④ 빈칸을 채우시오.

주어진 HTML / CSS

<div class="slider">
  <ul class="slide-list">
    <li class="slide">슬라이드 1</li>
    <li class="slide">슬라이드 2</li>
    <li class="slide">슬라이드 3</li>
  </ul>

  <button type="button" class="btn-next">다음</button>
</div>
.slider { width: 300px; overflow: hidden; }
.slide-list { display: flex; transition: transform 0.3s; }
.slide { flex: 0 0 300px; }

const slideList = document.querySelector('.slide-list');
const slides = document.querySelectorAll('.slide');
const btnPrev = document.querySelector('.btn-prev');
const btnNext = document.querySelector('.btn-next');

let currentIndex = 0;

    currentIndex = 0;
  }
  moveSlide();
});

btnPrev.addEventListener('click', function () {
  currentIndex--;
  // 첫 슬라이드에서 이전을 누르면 마지막으로

  }

문제 5. 모달(팝업) 창 HTML 작성형

배점 30점 · 권장 시간 8분

다음 JS가 정상 동작하도록 HTML을 작성하시오.
조건: 모달은 처음에 숨김 상태. "열기" 버튼 → 표시, X 버튼 또는 어두운 배경(dim) 클릭 → 닫힘. 모달 내부 클릭 시에는 닫히지 않아야 함.

주어진 JS

const openBtn = document.querySelector('.btn-open');
const modal = document.querySelector('.modal');
const closeBtn = modal.querySelector('.btn-close');

openBtn.addEventListener('click', function () {
  modal.classList.add('show');
});

closeBtn.addEventListener('click', function () {
  modal.classList.remove('show');
});

// dim(배경) 클릭 시 닫기 — 모달 내부 클릭은 제외
modal.addEventListener('click', function (e) {
  if (e.target === modal) {
    modal.classList.remove('show');
  }
});

주어진 CSS

.modal { display: none; position: fixed; inset: 0; background: rgba(0,0,0,.5); }
.modal.show { display: flex; justify-content: center; align-items: center; }
.modal-box { background: #fff; padding: 20px; width: 300px; }

▼ 완성 시 동작 예시 (직접 클릭해보세요 — 배경 클릭으로 닫기 확인)

모범답안 보기 
<button type="button" class="btn-open">열기</button>

<div class="modal">
  <div class="modal-box">
    <h2>모달 제목</h2>
    <p>모달 내용입니다.</p>
    <button type="button" class="btn-close">X</button>
  </div>
</div>
  • .modal이 dim(배경) 그 자체 — 내용은 반드시 .modal-box로 한 겹 감싸서 안에 배치
  • e.target === modal 판정이 핵심: 상자가 없으면 내부 클릭에도 닫혀버림 (감점 포인트)
  • X 버튼은 modal.querySelector('.btn-close')로 찾으므로 모달 내부에 위치
  • "처음에 숨김"은 CSS display: none이 처리 — HTML에 show를 붙이면 오답

문제 6. 폼 유효성 검사 오류 수정

배점 40점 · 권장 시간 10분

회원가입 폼 유효성 검사 코드에 오류가 있다. 모두 찾아 수정하시오.
동작: 이름이 비면 경고, 이메일에 @가 없으면 경고, 통과 시에만 "가입 완료". 오류 시 폼 전송 금지.

주어진 HTML

<form id="joinForm">
  <input type="text" id="userName" placeholder="이름">
  <p class="err-name"></p>
  <input type="text" id="userEmail" placeholder="이메일">
  <p class="err-email"></p>
  <button type="submit">가입하기</button>
</form>

오류가 포함된 JS

const form = document.getElementById('joinForm');
const userName = document.getElementById('userName');
const userEmail = document.getElementById('userEmail');
const errName = document.querySelector('.err-name');
const errEmail = document.querySelector('.err-email');

form.addEventListener('submit', function (e) {
  errName.textContent = '';
  errEmail.textContent = '';
  let isValid = true;

  if (userName.value.trim() === '') {
    errName.textContent = '이름을 입력하세요.';
    e.preventDefault;
    isValid = false;
  }

  if (userEmail.value.indexOf('@') === -1) {
    errEmail.innerText() = '이메일 형식이 아닙니다.';
    e.preventDefault();
    isValid = false;
  }

  if (isValid = true) {
    alert('가입 완료');
  }
});

▼ 완성 시 동작 예시 (빈 값 / @없는 이메일로 시도해보세요)

모범답안 보기 
form.addEventListener('submit', function (e) {
  errName.textContent = '';
  errEmail.textContent = '';
  let isValid = true;

  if (userName.value.trim() === '') {
    errName.textContent = '이름을 입력하세요.';
    e.preventDefault();                          // (1) 괄호 추가
    isValid = false;
  }

  if (userEmail.value.indexOf('@') === -1) {
    errEmail.textContent = '이메일 형식이 아닙니다.'; // (2)
    e.preventDefault();
    isValid = false;
  }

  if (isValid === true) {                       // (3)
    alert('가입 완료');
  }
});
  1. e.preventDefault; → e.preventDefault(); — 괄호가 없으면 참조만 하고 실행 안 됨 → 폼이 그대로 전송. 오류 수정 단골 1순위
  2. errEmail.innerText() = '...' → errEmail.textContent = '...' — innerText는 속성이지 함수가 아님(TypeError)
  3. isValid = true → isValid === true — 대입이라 항상 true → 실패해도 "가입 완료"가 뜸 (if (isValid)도 정답)
  4. 세트 함정: preventDefault()가 분기마다 있는지 확인하는 습관이 채점 기준

폼 검사 오류는 거의 항상 ① preventDefault 괄호 누락 ② textContent/innerText/innerHTML 혼동 ③ = vs === ④ trim() 누락/오타 중에서 출제됩니다.
